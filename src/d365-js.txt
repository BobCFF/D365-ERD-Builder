function metaValue(m,raw){ raw=(raw||'').trim(); if(raw==='')return '';
  if(m.kind==='bool')return /^(1|true|yes)$/i.test(raw)?'Yes':/^(0|false|no)$/i.test(raw)?'No':raw;
  if(m.key==='source')return ({'0':'Simple','1':'Calculated','2':'Rollup'})[raw]||raw;
  return raw; }
/* ---- in-browser parse of a Dataverse customizations.xml ---- */
function catFn(name,referenced,referencing,attr){
  const n=(name||'').toLowerCase(), a=(attr||'').toLowerCase(), rd=(referenced||'').toLowerCase();
  if(['createdby','modifiedby','createdonbehalfby','modifiedonbehalfby'].includes(a))return 'audit';
  if(['owninguser','owningteam','owningbusinessunit'].includes(a)||a.startsWith('owner'))return 'ownership';
  if(['businessunit','organization','team'].includes(rd)||/^(business_unit_|organization_|team_|owner_|lk_|user_)/.test(n))return 'ownership';
  if(/^(adx_|mspp_|msa_)/.test(n))return 'portal';
  const gen=new Set(['lead','opportunity','quote','invoice','salesorder','incident','site','campaign','campaignactivity','bulkoperation','sla','mailbox','service','bookableresource','bookableresourcebooking','bookableresourcebookingheader','knowledgebaserecord','knowledgearticle','interactionforemail','recurringappointmentmaster','activityparty','processstage','asyncoperation','workflow','transactioncurrency','fileattachment','adx_invitation','adx_inviteredemption','adx_portalcomment','approvalprocess','approvalstageapproval']);
  if(gen.has(rd))return 'generic';
  if(/^(activity_pointer_|transactioncurrency_|fileattachment_|processstage_|sla|bulkdelete|duplicate|principalobjectattribute|import)/.test(n))return 'generic';
  return 'business';
}

const SEC_PIE={none:{f:0}, user:{s:-0.125,f:0.25}, bu:{s:0.25,f:0.5}, deep:{s:0.125,f:0.75}, org:{f:1}};
function secPie(level){ const spec=SEC_PIE[level]||{f:0}, r=6.5, cx=8, cy=8, cls='sp-'+level;
  if(spec.f>=1)return `<svg class="spie ${cls}" viewBox="0 0 16 16" aria-hidden="true"><circle cx="8" cy="8" r="${r}"/></svg>`;
  let wedge='';
  if(spec.f>0){ const A=t=>[(cx+r*Math.sin(2*Math.PI*t)).toFixed(2),(cy-r*Math.cos(2*Math.PI*t)).toFixed(2)];
    const p0=A(spec.s), p1=A(spec.s+spec.f), large=spec.f>0.5?1:0;
    wedge=`<path d="M${cx} ${cy} L${p0[0]} ${p0[1]} A${r} ${r} 0 ${large} 1 ${p1[0]} ${p1[1]} Z"/>`; }
  return `<svg class="spie ${cls}" viewBox="0 0 16 16" aria-hidden="true"><circle class="rim" cx="8" cy="8" r="${r}" fill="none"/>${wedge}</svg>`; }

// a privilege name is prv<Verb><EntityLogicalName>; the remainder after the verb is the entity token
function parsePrivName(name){ if(!name||name.slice(0,3)!=='prv')return null; const body=name.slice(3);
  for(const [k,P] of SEC_VERBS)if(body.startsWith(P))return {verb:k, ent:body.slice(P.length).toLowerCase()};
  return null; }
// level is exported as a word (Basic/Local/Deep/Global) or a depth-mask number (1/2/4/8; 0 = none)
function parsePrivLevel(v){ v=(v||'').trim().toLowerCase();
  if(v==='1'||v==='basic'||v==='user')return 'user';
  if(v==='2'||v==='local'||v==='businessunit'||v==='bu')return 'bu';
  if(v==='4'||v==='deep'||v==='parentchild')return 'deep';
  if(v==='8'||v==='global'||v==='organization'||v==='org')return 'org';
  return 'none'; }
// a field-security permission (cancreate/canread/canupdate) is a numeric mask: 0 = not allowed, >0 = allowed
function fieldPermAllowed(v){ v=(v||'').trim(); return v!=='' && v!=='0' && v.toLowerCase()!=='false'; }
function parseCustomizationsXml(text){
  if(!text||!text.trim()) throw new Error('The file is empty.');
  const doc=new DOMParser().parseFromString(text,'application/xml');
  const perr=doc.querySelector('parsererror');
  if(perr){ const msg=(perr.textContent||'').replace(/\s+/g,' ').trim().slice(0,160); throw new Error('This isn’t valid XML'+(msg?' — '+msg:'.')); }
  if(!doc.documentElement) throw new Error('This isn’t valid XML (no root element).');
  if(doc.documentElement.tagName!=='ImportExportXml' && !doc.getElementsByTagName('Entities').length)
    throw new Error('This doesn’t look like a Dynamics 365 solution export — expected an <ImportExportXml> file (customizations.xml).');
  const ct=(el,tag)=>{ for(const c of el.children)if(c.tagName===tag)return (c.textContent||'').trim(); return ''; };
  const child=(el,tag)=>{ for(const c of el.children)if(c.tagName===tag)return c; return null; };
  const entities={};
  for(const ent of doc.getElementsByTagName('Entity')){
    const nameEl=child(ent,'Name'); if(!nameEl)continue;
    const logical=(nameEl.textContent||'').trim().toLowerCase(); if(!logical)continue;
    const disp=nameEl.getAttribute('LocalizedName')||nameEl.textContent.trim()||logical;
    const einfo=child(ent,'EntityInfo'), eentity=einfo&&child(einfo,'entity'), attrsEl=eentity&&child(eentity,'attributes');
    const attrs=[];
    if(attrsEl)for(const a of attrsEl.children){ if(a.tagName!=='attribute')continue;
      const an=(ct(a,'LogicalName')||ct(a,'Name')||'').toLowerCase(); if(!an)continue;
      const dn=a.querySelector('displaynames > displayname'), de=a.querySelector('Descriptions > Description');
      const at={name:an,type:ct(a,'Type'),required:ct(a,'RequiredLevel'),display:dn?dn.getAttribute('description')||'':'',desc:de?de.getAttribute('description')||'':''};
      for(const m of META_COLS){ const v=metaValue(m,ct(a,m.tag)); if(v!=='')at[m.key]=v; }
      attrs.push(at);
    }
    if(entities[logical]){ const seen=new Set(entities[logical].attributes.map(x=>x.name)); for(const x of attrs)if(!seen.has(x.name)){entities[logical].attributes.push(x);seen.add(x.name);} }
    else entities[logical]={logical,display:disp,attributes:attrs};
  }
  const rels=[], seen=new Set();
  for(const r of doc.getElementsByTagName('EntityRelationship')){
    const type=ct(r,'EntityRelationshipType'), referenced=ct(r,'ReferencedEntityName').toLowerCase(),
          referencing=ct(r,'ReferencingEntityName').toLowerCase(), attr=ct(r,'ReferencingAttributeName').toLowerCase();
    const de=r.querySelector('RelationshipDescription > Descriptions > Description'), label=de?de.getAttribute('description')||'':'';
    const name=r.getAttribute('Name')||'', key=referenced+'|'+referencing+'|'+attr+'|'+type;
    if(seen.has(key))continue; seen.add(key);
    rels.push({name,type,referenced,referencing,attr,label,cat:catFn(name,referenced,referencing,attr)});
  }
  for(const r of rels){ if(r.type!=='OneToMany')continue; const e=entities[r.referencing]; if(!e)continue;
    const a=e.attributes.find(x=>x.name===r.attr); if(a){ a.fk=true; a.targets=a.targets||[]; if(!a.targets.includes(r.referenced))a.targets.push(r.referenced); } }
  if(!Object.keys(entities).length) throw new Error('No tables found. A customizations.xml normally contains <Entity> definitions — this export may not include entity metadata.');
  // security roles: keep only privileges whose entity token matches a parsed table
  const roles=[], rolesEl=doc.getElementsByTagName('Roles')[0];
  if(rolesEl)for(const role of rolesEl.getElementsByTagName('Role')){
    const rname=(role.getAttribute('name')||role.getAttribute('id')||'').trim(); if(!rname)continue;
    const priv={};
    for(const rp of role.getElementsByTagName('RolePrivilege')){
      const pp=parsePrivName(rp.getAttribute('name')); if(!pp||!entities[pp.ent])continue;
      const lv=parsePrivLevel(rp.getAttribute('level')); if(lv==='none')continue;
      const cur=priv[pp.ent]||(priv[pp.ent]={});
      if(!cur[pp.verb]||SEC_LEVEL_RANK[lv]>SEC_LEVEL_RANK[cur[pp.verb]])cur[pp.verb]=lv;
    }
    roles.push({id:role.getAttribute('id')||'', name:rname, priv});
  }
  roles.sort((a,b)=>a.name.localeCompare(b.name));
  // field-level security: FieldSecurityProfiles → per-profile secured-column permissions
  const fsprofiles=[], fspEl=doc.getElementsByTagName('FieldSecurityProfiles')[0];
  if(fspEl)for(const prof of fspEl.getElementsByTagName('FieldSecurityProfile')){
    const pname=(prof.getAttribute('name')||prof.getAttribute('id')||'').trim(); if(!pname)continue;
    const perms=[];
    for(const fp of prof.getElementsByTagName('FieldPermission')){
      const ent=(fp.getAttribute('entityname')||'').toLowerCase(), attr=(fp.getAttribute('attributename')||'').toLowerCase();
      if(!ent||!attr)continue;
      perms.push({ent, attr, create:fieldPermAllowed(fp.getAttribute('cancreate')), read:fieldPermAllowed(fp.getAttribute('canread')), update:fieldPermAllowed(fp.getAttribute('canupdate'))});
    }
    fsprofiles.push({id:prof.getAttribute('id')||'', name:pname, perms});
  }
  fsprofiles.sort((a,b)=>a.name.localeCompare(b.name));
  return {entities,relationships:rels,roles,fsprofiles};
}
/* ---- IndexedDB storage for an imported schema (large capacity) ---- */
const IDB_NAME='tdp-erd-db', IDB_STORE='kv', IDB_KEY='import';
function idbOpen(){ return new Promise((res,rej)=>{ let r; try{ r=indexedDB.open(IDB_NAME,1); }catch(e){ return rej(e); }
  r.onupgradeneeded=()=>{ try{ if(!r.result.objectStoreNames.contains(IDB_STORE))r.result.createObjectStore(IDB_STORE); }catch(e){} };
  r.onsuccess=()=>res(r.result); r.onerror=()=>rej(r.error||new Error('idb open failed')); r.onblocked=()=>rej(new Error('idb blocked')); }); }
function idbGet(){ return idbOpen().then(db=>new Promise((res,rej)=>{ const t=db.transaction(IDB_STORE,'readonly'); const g=t.objectStore(IDB_STORE).get(IDB_KEY); g.onsuccess=()=>res(g.result||null); g.onerror=()=>rej(g.error); })); }
function idbSet(v){ return idbOpen().then(db=>new Promise((res,rej)=>{ const t=db.transaction(IDB_STORE,'readwrite'); t.objectStore(IDB_STORE).put(v,IDB_KEY); t.oncomplete=()=>res(); t.onerror=()=>rej(t.error||new Error('idb write failed')); })); }
function idbDel(){ return idbOpen().then(db=>new Promise((res,rej)=>{ const t=db.transaction(IDB_STORE,'readwrite'); t.objectStore(IDB_STORE).delete(IDB_KEY); t.oncomplete=()=>res(); t.onerror=()=>rej(t.error); })); }

/* markers — crow's-foot (many, at child) + bar (one, at parent) */
// Markers reference point (refX,refY) = (0,0) sits at the path endpoint, which the
// edge builder insets INWARD from the box by MK_* so the line begins/ends at the
// marker apex. The glyphs are drawn back toward the box: at marker-start orient's +x
// points into the diagram, so the crow's foot spreads at -x (the box); at marker-end
// +x points into the box, so the "one" glyph sits at +x. Line never overlaps the feet.
const MK_CF=13, MK_ONEM=12, MK_ONEO=14, MK_DOT=12, MK_DIA=15;
function markerDefs(prefix, colf, bg){
  let s='';
  // scale marker line-work with the active connector weight so glyphs stay proportional
  const w=n=>(n*(lwPx()/1.4)).toFixed(2);
  for(const [cat,,v] of REL_CATS){ const c=colf(v,cat);
    // crow's foot (many) at the child end — feet toward the box (-x), apex at the endpoint
    s+=`<marker id="${prefix}-cf-${cat}" markerUnits="userSpaceOnUse" markerWidth="14" markerHeight="14" viewBox="-13 -7 14 14" refX="0" refY="0" orient="auto"><path d="M-12,-5 L0,0 M-12,5 L0,0 M-12,0 L0,0" fill="none" stroke="${c}" stroke-width="${w(1.3)}"/></marker>`;
    // mandatory one (exactly one): double bar toward the box (+x)
    s+=`<marker id="${prefix}-onem-${cat}" markerUnits="userSpaceOnUse" markerWidth="13" markerHeight="14" viewBox="0 -7 13 14" refX="0" refY="0" orient="auto"><path d="M5,-5.5 L5,5.5 M9,-5.5 L9,5.5" stroke="${c}" stroke-width="${w(1.6)}" fill="none"/></marker>`;
    // optional one (zero or one): bar + circle toward the box (+x)
    s+=`<marker id="${prefix}-oneo-${cat}" markerUnits="userSpaceOnUse" markerWidth="15" markerHeight="14" viewBox="0 -7 15 14" refX="0" refY="0" orient="auto"><path d="M4,-5.5 L4,5.5" stroke="${c}" stroke-width="${w(1.6)}" fill="none"/><circle cx="10" cy="0" r="3.1" fill="${bg}" stroke="${c}" stroke-width="${w(1.3)}"/></marker>`;
    // IDEF1X: filled dot (the "many"/child cardinality) at the child end, toward the box (-x)
    s+=`<marker id="${prefix}-dot-${cat}" markerUnits="userSpaceOnUse" markerWidth="14" markerHeight="14" viewBox="-13 -7 14 14" refX="0" refY="0" orient="auto"><circle cx="-6" cy="0" r="3.4" fill="${c}" stroke="none"/></marker>`;
    // IDEF1X: hollow diamond at the parent end (only for an OPTIONAL/nullable relationship), toward the box (+x)
    s+=`<marker id="${prefix}-dia-${cat}" markerUnits="userSpaceOnUse" markerWidth="16" markerHeight="14" viewBox="0 -7 16 14" refX="0" refY="0" orient="auto"><path d="M2.5,0 L7.5,-4.6 L12.5,0 L7.5,4.6 Z" fill="${bg}" stroke="${c}" stroke-width="${w(1.3)}"/></marker>`;
  }
  return s;
}

// Whether a connector is drawn dashed. In IDEF1X the line style carries the

// a point on a quadratic at parameter t
function qpt(p1,c,p2,t){ const u=1-t; return [u*u*p1[0]+2*u*t*c[0]+t*t*p2[0], u*u*p1[1]+2*u*t*c[1]+t*t*p2[1]]; }
function cpt(p1,c1,c2,p2,t){ const u=1-t; return [u*u*u*p1[0]+3*u*u*t*c1[0]+3*u*t*t*c2[0]+t*t*t*p2[0], u*u*u*p1[1]+3*u*u*t*c1[1]+3*u*t*t*c2[1]+t*t*t*p2[1]]; }
const _u=(x,y)=>{ const l=Math.hypot(x,y)||1; return [x/l,y/l]; };
// Pull the drawn line's endpoints in from the box (toward each control point) by the
// marker length, so the crow's-foot / one markers sit in the gap without the line
// crossing them. No-op in UML mode (no glyphs). t1/t2 are the control points.
function insetEnds(p1,t1,p2,t2,mand){ if(notation==='uml')return[p1,p2];
  const room=Math.hypot(p2[0]-p1[0],p2[1]-p1[1]);
  // child-end glyph length, and parent-end glyph length (IDEF1X shows a parent
  // glyph only for optional relationships; mandatory has none, so no gap there)
  const cf=notation==='idef1x'?MK_DOT:MK_CF;
  const one=notation==='idef1x'?(mand?0:MK_DIA):(mand?MK_ONEM:MK_ONEO);
  const f=Math.min(1, room*0.6/((cf+one)||1));
  const u1=_u(t1[0]-p1[0],t1[1]-p1[1]), u2=_u(t2[0]-p2[0],t2[1]-p2[1]);
  return [[p1[0]+u1[0]*cf*f, p1[1]+u1[1]*cf*f],[p2[0]+u2[0]*one*f, p2[1]+u2[1]*one*f]]; }
function buildEdges(edges){
  const out=[];
  const nonSelf=edges.filter(r=>r.referenced!==r.referencing);
  const selfs=edges.filter(r=>r.referenced===r.referencing);
  if(lineMode==='s'){
    // Give every connector its own slot along a box's left/right edge so they
    // don't overlap where they meet the box. Two passes: collect, then place.
    const slots={}; // nodeId -> {L:[],R:[]}
    const recs=nonSelf.map(r=>{ const a=N[r.referencing], b=N[r.referenced];
      const childRight=(b.x+b.w/2)>=(a.x+a.w/2);
      const rec={r,a,b,childRight,aSide:childRight?'R':'L',bSide:childRight?'L':'R'};
      (slots[r.referencing]=slots[r.referencing]||{L:[],R:[]})[rec.aSide].push({rec,end:'a',oy:b.y+b.h/2});
      (slots[r.referenced]=slots[r.referenced]||{L:[],R:[]})[rec.bSide].push({rec,end:'b',oy:a.y+a.h/2});
      return rec; });
    for(const id in slots){ const n=N[id]; for(const side of ['L','R']){ const L=slots[id][side]; if(!L.length)continue;
      L.sort((p,q)=>p.oy-q.oy);                                  // order by the far end to reduce crossings
      const pad=Math.min(16,n.h*0.22), span=Math.max(0,n.h-2*pad), m=L.length, x=side==='R'?n.x+n.w:n.x;
      L.forEach((it,i)=>{ const y=m===1?n.y+n.h/2:n.y+pad+span*((i+0.5)/m); it.rec[it.end==='a'?'p1':'p2']=[x,y]; }); } }
    for(const rec of recs){ const p1=rec.p1,p2=rec.p2, dir=rec.childRight?1:-1, kk=clamp(Math.abs(p2[0]-p1[0])*0.5,44,200);
      const c1=[p1[0]+dir*kk,p1[1]], c2=[p2[0]-dir*kk,p2[1]];
      const [s1,s2]=insetEnds(p1,c1,p2,c2,rec.r.mandatory);
      const d=`M${s1[0]},${s1[1]} C${c1[0]},${c1[1]} ${c2[0]},${c2[1]} ${s2[0]},${s2[1]}`;
      out.push({r:rec.r,d,p1:s1,p2:s2,star:cpt(s1,c1,c2,s2,0.12),one:cpt(s1,c1,c2,s2,0.88),mid:cpt(s1,c1,c2,s2,0.5)}); }
  } else {
    const pairTot={},seen={};
    for(const r of nonSelf){ const k=[r.referenced,r.referencing].sort().join('|'); pairTot[k]=(pairTot[k]||0)+1; }
    for(const r of nonSelf){ const a=N[r.referencing],b=N[r.referenced];
      const k=[r.referenced,r.referencing].sort().join('|'); const idx=(seen[k]=(seen[k]||0)+1)-1, tot=pairTot[k];
      const acx=a.x+a.w/2,acy=a.y+a.h/2,bcx=b.x+b.w/2,bcy=b.y+b.h/2;
      const p1=anchor(a,bcx,bcy),p2=anchor(b,acx,acy);
      const midx=(p1[0]+p2[0])/2,midy=(p1[1]+p2[1])/2;
      let nx=-(p2[1]-p1[1]),ny=(p2[0]-p1[0]); const len=Math.hypot(nx,ny)||1; nx/=len;ny/=len;
      let off=(idx-(tot-1)/2)*26; if(lineMode==='curved') off+=clamp(len*0.16,22,110);
      const c=[midx+nx*off, midy+ny*off];
      const [s1,s2]=insetEnds(p1,c,p2,c,r.mandatory);
      const d=`M${s1[0]},${s1[1]} Q${c[0]},${c[1]} ${s2[0]},${s2[1]}`;
      out.push({r,d,p1:s1,p2:s2,star:qpt(s1,c,s2,0.14),one:qpt(s1,c,s2,0.86),mid:qpt(s1,c,s2,0.5)}); }
  }
  const selfSeen={};
  for(const r of selfs){ const a=N[r.referencing]; const idx=(selfSeen[r.referenced]=(selfSeen[r.referenced]||0)+1)-1;
    // A horseshoe (sideways "U") opening toward the box: the two tips leave the edge
    // HORIZONTALLY (control points level with each endpoint), so the crow's-foot / one
    // markers sit clear of the box and point straight into it — instead of the pinched
    // teardrop you get when the control points sit far above/below the endpoints.
    const x=a.x+a.w, y=a.y+a.h*0.5, h=20+idx*11, off=52+idx*30,
      p1=[x,y-h], p2=[x,y+h], c1=[x+off,y-h], c2=[x+off,y+h];
    const [s1,s2]=insetEnds(p1,c1,p2,c2,r.mandatory);
    const d=`M${s1[0]},${s1[1]} C${c1[0]},${c1[1]} ${c2[0]},${c2[1]} ${s2[0]},${s2[1]}`;
    out.push({r,d,p1:s1,p2:s2,star:cpt(s1,c1,c2,s2,0.22),one:cpt(s1,c1,c2,s2,0.78),mid:cpt(s1,c1,c2,s2,0.5)}); }
  return out;
}

function forceLayout(ids,fixed){
  if(!ids.length)return;
  const cols=Math.max(1,Math.ceil(Math.sqrt(ids.length))), rows=Math.ceil(ids.length/cols);
  const CX=340, CY=250;
  const P=ids.map((id,i)=>{ const n=N[id]; const saved=fixed&&fixed.has(id);
    let x,y;
    if(saved){ x=n.x+n.w/2; y=n.y+n.h/2; }
    else { x=((i%cols)-(cols-1)/2)*CX + (Math.random()*60-30); y=(Math.floor(i/cols)-(rows-1)/2)*CY + (Math.random()*60-30); }
    return {id,x,y,w:n.w,h:n.h,vx:0,vy:0}; });
  const idx=Object.fromEntries(P.map((p,i)=>[p.id,i]));
  const links=edgeList().filter(r=>idx[r.referenced]!=null&&idx[r.referencing]!=null&&r.referenced!==r.referencing);
  for(let it=0;it<360;it++){
    const alpha=1-it/360;
    for(let i=0;i<P.length;i++)for(let j=i+1;j<P.length;j++){ const a=P[i],b=P[j]; let dx=a.x-b.x,dy=a.y-b.y; let d2=dx*dx+dy*dy||0.01; let d=Math.sqrt(d2);
      let rep=180000/d2; if(rep>1600)rep=1600; let fx=dx/d*rep,fy=dy/d*rep; a.vx+=fx;a.vy+=fy;b.vx-=fx;b.vy-=fy; }
    for(const l of links){ const a=P[idx[l.referencing]],b=P[idx[l.referenced]]; let dx=b.x-a.x,dy=b.y-a.y; let d=Math.hypot(dx,dy)||1; let f=(d-260)*0.035; let fx=dx/d*f,fy=dy/d*f; a.vx+=fx;a.vy+=fy;b.vx-=fx;b.vy-=fy; }
    for(const p of P){ p.vx-=p.x*0.006; p.vy-=p.y*0.006; }
    for(const p of P){ if(fixed&&fixed.has(p.id)){p.vx=p.vy=0;continue;} p.x+=clamp(p.vx*alpha,-40,40); p.y+=clamp(p.vy*alpha,-40,40); p.vx*=0.85;p.vy*=0.85; }
  }
  for(let pass=0;pass<14;pass++)for(let i=0;i<P.length;i++)for(let j=i+1;j<P.length;j++){ const a=P[i],b=P[j];
    const dx=a.x-b.x,dy=a.y-b.y; const ox=(a.w+b.w)/2+26-Math.abs(dx), oy=(a.h+b.h)/2+18-Math.abs(dy);
    if(ox>0&&oy>0){ if(ox<oy){ const s=(dx<0?-1:1)*ox/2; if(!(fixed&&fixed.has(a.id)))a.x+=s; if(!(fixed&&fixed.has(b.id)))b.x-=s; }
      else { const s=(dy<0?-1:1)*oy/2; if(!(fixed&&fixed.has(a.id)))a.y+=s; if(!(fixed&&fixed.has(b.id)))b.y-=s; } } }
  for(const p of P){ N[p.id].x=p.x-p.w/2; N[p.id].y=p.y-p.h/2; place(p.id); }
}

async function saveFile(filename,data){

    if(dl){ await dl.save({filename,data}); return 'saved'; } }

  const url=URL.createObjectURL(blob); const a=document.createElement('a'); a.href=url; a.download=filename;
  document.body.appendChild(a); a.click(); a.remove(); setTimeout(()=>URL.revokeObjectURL(url),4000);

// One-pass aggregation of the parsed model into report-ready metrics. Recomputed
// on each view entry so it always reflects the currently loaded schema.
function computeInsights(){
  let cols=0, custF=0, auditF=0, secF=0, reqF=0, withDisp=0, withDesc=0;
  const typeCount={}, reqCount={}, perTable=[];
  for(const id of allIds){
    const e=ENT[id], n=e.attributes.length; cols+=n; perTable.push({id,n});
    for(const a of e.attributes){
      const t=(a.type||'—').trim()||'—'; typeCount[t]=(typeCount[t]||0)+1;
      const rq=(a.required||'none').toLowerCase(); reqCount[rq]=(reqCount[rq]||0)+1;
      if(rq==='required'||rq==='systemrequired'||rq==='applicationrequired')reqF++;
      if(a.custom==='Yes')custF++;
      if(a.audit==='Yes')auditF++;
      if(a.secured==='Yes')secF++;
      if(a.display&&a.display.trim())withDisp++;
      if(a.desc&&a.desc.trim())withDesc++;
    }
  }
  const custTables=allIds.filter(id=>ENT[id].group!=='__std');
  // relationships by category (all parsed rels carry a .cat)
  const relCat={}; for(const r of RELS)relCat[r.cat]=(relCat[r.cat]||0)+1;
  const relType={}; for(const r of RELS)relType[r.type||'—']=(relType[r.type||'—']||0)+1;
  const selfRefs=RELS.filter(r=>r.referenced===r.referencing).length;
  // group (publisher-prefix) breakdown, largest first, mirroring GROUPS order
  const byGroup=GROUPS.map(([pf,label])=>({pf,label,color:GVAR[pf],n:allIds.filter(id=>ENT[id].group===pf).length})).filter(g=>g.n>0);
  const topBySize=perTable.slice().sort((a,b)=>b.n-a.n||(ENT[a.id].display||a.id).localeCompare(ENT[b.id].display||b.id)).slice(0,8);
  const connected=allIds.map(id=>({id,n:bizCount[id]||0})).sort((a,b)=>b.n-a.n).filter(x=>x.n>0).slice(0,8);
  const isolated=allIds.filter(id=>!bizCount[id]);
  return {cols,custF,auditF,secF,reqF,withDisp,withDesc,typeCount,reqCount,relCat,relType,selfRefs,
    byGroup,topBySize,connected,isolated,custTables};
}
const REQ_LABEL={none:'Optional',recommended:'Business recommended',required:'Business required',systemrequired:'System required',applicationrequired:'Application required'};
function insBars(rows,total,fmt){ // rows: [{label,n,color?,dot?}]
  const max=Math.max(1,...rows.map(r=>r.n));
  return `<div class="bars">`+rows.map(r=>{
    const w=Math.round((r.n/max)*100), col=r.color?`var(${r.color})`:'var(--accent)';
    const dot=r.dot?`<span class="dot" style="background:${col}"></span>`:'';
    const val=fmt?fmt(r.n):String(r.n);
    return `<div class="bar-row"><span class="b-lbl">${dot}<span class="t" title="${esc(r.label)}">${esc(r.label)}</span></span>`+
      `<span class="bar-track"><span class="bar-fill" style="width:${w}%;background:${col}"></span></span>`+
      `<span class="b-val">${esc(val)}</span></div>`; }).join('')+`</div>`;
}

}

function colPass(d,skipEnt){ const F=dcolF;
  if(!skipEnt){ // the per-table dialog hides the Table column, so its filter doesn't apply there
    if(F.ent.groups.length&&!F.ent.groups.includes(d.group))return false;
    if(F.ent.tables.length&&!F.ent.tables.includes(d.ent))return false;
    if(F.ent.q&&!((d.entdisp+' '+d.ent).toLowerCase().includes(F.ent.q.toLowerCase())))return false; }
  if(F.name.q&&!d.name.toLowerCase().includes(F.name.q.toLowerCase()))return false;
  if(!matchBlank(F.display,d.display))return false;
  if(F.type.sel.length&&!F.type.sel.includes(d.type))return false;
  if(F.req.mode==='req'&&!d.req)return false; if(F.req.mode==='opt'&&d.req)return false;
  if(!matchKey(F.key.mode,d))return false;
  if(!matchBlank(F.refs,d.refs))return false;
  if(!matchBlank(F.desc,d.desc))return false;
  for(const mk of META_KEYS){ const ff=F[mk]; if(!ff)continue; const mv=d[mk]!=null?String(d[mk]):'';
    if(DCOLDEF[mk].filter==='boolsel'){ if(ff.mode==='yes'&&mv!=='Yes')return false; if(ff.mode==='no'&&mv!=='No')return false; }
    else if(!matchBlank(ff,mv))return false; }
  return true; }

const _CRC=(()=>{ const t=new Uint32Array(256); for(let n=0;n<256;n++){ let c=n; for(let k=0;k<8;k++)c=(c&1)?(0xEDB88320^(c>>>1)):(c>>>1); t[n]=c>>>0; } return t; })();
function crc32(buf){ let c=0xFFFFFFFF; for(let i=0;i<buf.length;i++)c=_CRC[(c^buf[i])&0xFF]^(c>>>8); return (c^0xFFFFFFFF)>>>0; }
function zipStore(files){ const parts=[],central=[]; let offset=0;
  const u16=n=>[n&255,(n>>8)&255], u32=n=>[n&255,(n>>8)&255,(n>>16)&255,(n>>24)&255];
  for(const f of files){ const nameB=new TextEncoder().encode(f.name), data=f.data, crc=crc32(data);
    const lh=[].concat([0x50,0x4b,0x03,0x04],u16(20),u16(0),u16(0),u16(0),u16(0),u32(crc),u32(data.length),u32(data.length),u16(nameB.length),u16(0));
    parts.push(new Uint8Array(lh),nameB,data); central.push({crc,len:data.length,nameB,offset}); offset+=lh.length+nameB.length+data.length; }
  const cd=[]; let cdSize=0;
  for(const c of central){ const h=[].concat([0x50,0x4b,0x01,0x02],u16(20),u16(20),u16(0),u16(0),u16(0),u16(0),u32(c.crc),u32(c.len),u32(c.len),u16(c.nameB.length),u16(0),u16(0),u16(0),u16(0),u32(0),u32(c.offset));
    const arr=new Uint8Array(h.length+c.nameB.length); arr.set(h,0); arr.set(c.nameB,h.length); cd.push(arr); cdSize+=arr.length; }
  const end=[].concat([0x50,0x4b,0x05,0x06],u16(0),u16(0),u16(central.length),u16(central.length),u32(cdSize),u32(offset),u16(0));
  const all=[...parts,...cd,new Uint8Array(end)]; let total=all.reduce((s,p)=>s+p.length,0);
  const out=new Uint8Array(total); let o=0; for(const p of all){ out.set(p,o); o+=p.length; } return out; }

  const scope=(card.querySelector('input[name="ex-scope"]:checked')||{}).value||'filter';
  const cols=XCOLS.filter(c=>card.querySelector(`input[data-col="${c.id}"]`)?.checked);
  if(!cols.length){ toast('Pick at least one column.'); return; }
  exportOpts={fmt,layout,scope:(scope==='entity'?exportOpts.scope:scope),cols:cols.map(c=>c.id)}; saveState();

  const base = ent?`d365-dictionary-${ent}`:'d365-data-dictionary';
  closeModal();
  if(fmt==='csv'){
    const lines=[head.map(csvCell).join(',')];
    for(const d of rows)lines.push(cols.map(c=>csvCell(c.get(d))).join(','));
    saveExport(`${base}-${stamp}.csv`, '﻿'+lines.join('\r\n'), `${rows.length} rows`);
  } else {
    let sheets;
    if(layout==='single' || ent){ sheets=[{name: ent?xlsxSheetName(ENT[ent].display||ent,new Set()):'Data dictionary', rows:[head,...rows.map(d=>cols.map(c=>c.get(d)))]}]; }
    else {
      const used=new Set(), byEnt=new Map();
      for(const d of rows){ if(!byEnt.has(d.ent))byEnt.set(d.ent,[]); byEnt.get(d.ent).push(d); }
      const ohead=['Table','Table (logical)','Group','Columns'];
      const orows=[...byEnt.entries()].map(([id,rs])=>[ (ENT[id]||{}).display||id, id, GLABEL[(ENT[id]||{}).group]||'', rs.length ]);
      sheets=[{name:xlsxSheetName('Overview',used), rows:[ohead,...orows]}];
      for(const [id,rs] of byEnt)sheets.push({name:xlsxSheetName((ENT[id]||{}).display||id,used), rows:[head,...rs.map(d=>cols.map(c=>c.get(d)))]});
    }
    let bytes; try{ bytes=buildXlsx(sheets); }catch(e){ toast('Excel build failed: '+(e&&e.message||e)); return; }
    saveExport(`${base}-${stamp}.xlsx`, bytes, `${sheets.length} sheet${sheets.length===1?'':'s'}`);
  }
}
